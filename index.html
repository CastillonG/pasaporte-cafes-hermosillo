}; // Detener cámara function stopCamera() { if (currentStream) { currentStream.getTracks().forEach(track => { track.stop(); }); currentStream = null; } const videoElement = document.getElementById('qr-video'); if (videoElement) { videoElement.srcObject = null; videoElement.remove(); } } // Iniciar cámara en vivo window.startCamera = async function() { const statusEl = document.getElementById('qr-status'); if (!statusEl) return; statusEl.textContent = '📷 Solicitando acceso a la cámara...'; statusEl.style.color = 'var(--espresso)'; try { stopCamera(); let videoElement = document.getElementById('qr-video'); if (!videoElement) { videoElement = document.createElement('video'); statusEl.innerHTML = '❌ No se pudo acceder a la cámara.
Usa la opción de código manual o carga una foto.'; statusEl.style.color = '#c0392b'; } }; // Escaneo continuo de QR function startQRScanning(videoElement, statusEl) { const canvas = document.createElement('canvas'); const context = canvas.getContext('2d'); let scanning = true; let lastScan = 0; function scanFrame() { if (!scanning) return; if (!videoElement.videoWidth || !videoElement.videoHeight) { requestAnimationFrame(scanFrame); return; } if (videoElement.readyState === videoElement.HAVE_ENOUGH_DATA) { canvas.width = videoElement.videoWidth; canvas.height = videoElement.videoHeight; context.drawImage(videoElement, 0, 0, canvas.width, canvas.height); try { const imageData = context.getImageData(0, 0, canvas.width, canvas.height); const code = jsQR(imageData.data, imageData.width, imageData.height, { inversionAttempts: "attemptBoth", }); if (code && code.data && (Date.now() - lastScan) > 1000) { lastScan = Date.now(); const qrData = code.data.trim().toLowerCase(); console.log('QR detectado:', qrData); let foundCafe = null; // Buscar coincidencia con nombres de café for (let cafe of cafes) { const cafeNombre = cafe.name.toLowerCase(); if (qrData === cafeNombre || qrData.includes(cafeNombre) || (qrData === 'cafe26' && currentCafe)) { foundCafe = currentCafe || cafe; break; } } if (foundCafe) { statusEl.innerHTML = '✅ ¡QR válido!
Registrando visita...'; statusEl.style.color = 'var(--green)'; scanning = false; stopCamera(); setTimeout(() => { currentCafe = foundCafe; window.registrarVisita(); }, 500); return; } else { statusEl.innerHTML = '⚠️ QR no reconocido.
Solicita el código oficial al café'; statusEl.style.color = '#c0392b'; setTimeout(() => { if (statusEl && scanning) { statusEl.innerHTML = '🔍 Escaneando... enfoca el código QR'; statusEl.style.color = 'var(--espresso)'; } }, 2000); } } } catch (e) { console.error('Error al escanear:', e); } } if (scanning) { requestAnimationFrame(scanFrame); } } scanFrame(); } // Escanear desde foto window.scanQRFromPhoto = function(input) { const file = input.files[0]; if (!file) return; const statusEl = document.getElementById('qr-status'); statusEl.textContent = '🔍 Analizando QR desde imagen...'; statusEl.style.color = 'var(--espresso)'; const img = new Image(); const url = URL.createObjectURL(file); img.onload = function() { const canvas = document.getElementById('qr-canvas'); const ctx = canvas.getContext('2d'); canvas.width = img.width; canvas.height = img.height; ctx.drawImage(img, 0, 0); URL.revokeObjectURL(url); try { const imageData = ctx.getImageData(0, 0, canvas.width, canvas.height); const code = jsQR(imageData.data, imageData.width, imageData.height, { inversionAttempts: 'attemptBoth' }); if (code && code.data) { const qrData = code.data.trim().toLowerCase(); let foundCafe = null; for (let cafe of cafes) { if (qrData === cafe.name.toLowerCase() || qrData.includes(cafe.name.toLowerCase()) || qrData === 'cafe26') { foundCafe = cafe; break; } } if (foundCafe || (qrData === 'cafe26' && currentCafe)) { statusEl.innerHTML = '✅ ¡QR válido! Registrando...'; statusEl.style.color = 'var(--green)'; setTimeout(() => { if (foundCafe) currentCafe = foundCafe; window.registrarVisita(); input.value = ''; }, 700); } else { statusEl.innerHTML = '❌ QR no válido para ningún café registrado'; statusEl.style.color = '#c0392b'; input.value = ''; } } else { statusEl.innerHTML = '❌ No se detectó ningún QR válido en la imagen'; statusEl.style.color = '#c0392b'; input.value = ''; } } catch(e) { statusEl.innerHTML = '❌ Error al leer la imagen'; statusEl.style.color = '#c0392b'; input.value = ''; } }; img.onerror = function() { statusEl.innerHTML = '❌ Error al cargar la imagen'; statusEl.style.color = '#c0392b'; input.value = ''; }; img.src = url; }; // Funciones de UI window.showCamera = function() { stopCamera(); document.getElementById('verify-choose').style.display = 'none'; document.getElementById('verify-camera').style.display = 'block';
